ระบบเช็กรายการรับผ้าคาด

กำลังโหลดข้อมูลจากชีท...

พิมพ์รหัสนักศึกษาหรือชื่อ แล้วกด "ค้นหา" เพื่อตรวจสอบข้อมูล

สถานะการบันทึก: บันทึกในเครื่อง (Local Storage)

เชื่อมต่อ Google Sheet "รับผ้าคาด"

หากเว้นว่างไว้ ระบบจะบันทึกสถานะการรับของไว้ใน Local Storage ของเบราว์เซอร์ชั่วคราว

โค้ด Google Apps Script สั้นๆ (สำหรับชีท "รับผ้าคาด")

คัดลอกโค้ดนี้ไปวางใน Google Sheet -> ส่วนขยาย (Extensions) -> Apps Script แล้วกด "นำไปใช้งานเป็น Web App":

function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName("รับผ้าคาด");
    
    // ถ้ายังไม่มีชีท "รับผ้าคาด" ให้สร้างให้อัตโนมัติ
    if (!sheet) {
      sheet = ss.insertSheet("รับผ้าคาด");
      sheet.appendRow(["วันเวลาบันทึก", "รหัสนักศึกษา", "ชื่อ-นามสกุล", "รายละเอียดอื่น"]);
    }
    
    // เพิ่มแถวบันทึกข้อมูลการรับ
    sheet.appendRow([
      new Date(),
      data.studentId || '',
      data.name || '',
      data.details || ''
    ]);
    
    return ContentService.createTextOutput(JSON.stringify({result: "success"}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({result: "error", message: err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
